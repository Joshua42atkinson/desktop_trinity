
fn spawn_weighed_words(
    mut commands: Commands,
    mut events: EventReader<WeighResult>,
) {
    for res in events.read() {
        if let Some(physics) = &res.physics {
            info!("✨ Spawning Weighed Word: {}", physics.word);
            commands.spawn((
                VocabularyItem {
                    word: physics.word.clone(),
                    definition: physics.definition.clone(),
                    tier: physics.tier,
                    tags: physics.tags.clone(),
                },
                CognitiveWeight {
                    base_mass: physics.mass,
                    effective_mass: physics.mass,
                },
            ));
        }
    }
}
use crate::game::vaam::{CognitiveWeight, VocabularyItem};
use bevy::{prelude::*, tasks::{AsyncComputeTaskPool, Task}};
use futures_lite::future;
use iron_road_physics::{CognitiveLoad, VocabularyTier};
use serde::{Deserialize, Serialize};
use serde_json::json;
use std::sync::{Arc, Mutex};

pub struct WeighStationPlugin;

impl Plugin for WeighStationPlugin {
    fn build(&self, app: &mut App) {
        app.add_event::<WeighRequest>()
            .add_event::<WeighResult>()
            .add_systems(
                Update,
                (
                    handle_weigh_requests,
                    handle_weigh_results,
                    spawn_weighed_words,
                ),
            );
    }
}

/// Event to trigger measuring a word
#[derive(Event)]
pub struct WeighRequest {
    pub word: String,
    // Optional: callback entity or location?
}

/// Event when measurement is complete
#[derive(Event)]
pub struct WeighResult {
    pub word: String,
    pub physics: Option<WordPhysics>,
}

/// The schema that the LLM (Llama 4 Scout) must adhere to when "weighing" a word.
#[derive(Debug, Serialize, Deserialize, Clone)]
pub struct WordPhysics {
    pub word: String,
    pub definition: String,
    pub tier: VocabularyTier,
    /// 1-100 (Intrinsic Load)
    pub mass: f32,
    /// Semantic tags for socket compatibility (e.g., ["Time", "Decay"])
    pub tags: Vec<String>,
}

#[derive(Component)]
pub struct WeighTask(Task<Option<WordPhysics>>);

fn handle_weigh_requests(
    mut commands: Commands,
    mut events: EventReader<WeighRequest>,
) {
    let thread_pool = AsyncComputeTaskPool::get();
    
    for req in events.read() {
        let word = req.word.clone();
        info!("⚖️ Weigh Station Impulse: '{}'", word);
        
        // Spawn async task
        let task = thread_pool.spawn(async move {
            weigh_word_async(&word).await
        });
        
        commands.spawn(WeighTask(task));
    }
}

fn handle_weigh_results(
    mut commands: Commands,
    mut query: Query<(Entity, &mut WeighTask)>,
    mut events: EventWriter<WeighResult>,
) {
    for (entity, mut task) in query.iter_mut() {
        if let Some(physics_opt) = future::block_on(future::poll_once(&mut task.0)) {
            // Task finished
            if let Some(physics) = &physics_opt {
                info!("📦 Received Physics for '{}': Mass={}", physics.word, physics.mass);
                events.send(WeighResult {
                    word: physics.word.clone(),
                    physics: Some(physics.clone()),
                });
            } else {
                warn!("⚠️ Failed to weigh word.");
                 events.send(WeighResult {
                    word: "Unknown".to_string(), // Should pass original word if possible, but task consumes it
                    physics: None,
                });
            }
            
            // Cleanup task entity
            commands.entity(entity).despawn();
        }
    }
}

/// Actual async implementation
async fn weigh_word_async(word: &str) -> Option<WordPhysics> {
    let client = reqwest::Client::new();
    // Assuming LM Studio / OpenAI Compatible local endpoint
    let url = "http://localhost:1234/v1/chat/completions"; 
    
    let schema_str = r#"
    {
        "type": "object",
        "properties": {
            "word": { "type": "string" },
            "definition": { "type": "string" },
            "tier": { "type": "string", "enum": ["Basic", "Academic", "Hazardous"] },
            "mass": { "type": "number" },
            "tags": { "type": "array", "items": { "type": "string" } }
        },
        "required": ["word", "definition", "tier", "mass", "tags"]
    }
    "#;

    let system_prompt = format!(
        "You are the Weigh Station. Analyze the word. Return JSON only conforming to: {}. Measure 'mass' (1-100) based on cognitive complexity.",
        schema_str
    );

    let payload = json!({
        "model": "local-model", // LM Studio ignores this usually
        "messages": [
            { "role": "system", "content": system_prompt },
            { "role": "user", "content": format!("Weigh this word: {}", word) }
        ],
        "temperature": 0.1,
        "max_tokens": 500,
        "response_format": { "type": "json_object" } // OpenAI compatible JSON mode
    });

    match client.post(url).json(&payload).send().await {
        Ok(resp) => {
            if let Ok(json) = resp.json::<serde_json::Value>().await {
                if let Some(content_str) = json["choices"][0]["message"]["content"].as_str() {
                    // Parse the inner JSON
                    if let Ok(physics) = serde_json::from_str::<WordPhysics>(content_str) {
                         return Some(physics);
                    } else {
                        warn!("Failed to parse inner JSON: {}", content_str);
                    }
                }
            } else {
                warn!("Failed to parse response JSON");
            }
        }
        Err(e) => {
            warn!("HTTP Request Failed: {}", e);
            // Fallback for offline dev
            warn!("🔌 Offline Mode: Returning Mock Data");
            return Some(WordPhysics {
                word: word.to_string(),
                definition: format!("(Offline) Definition for {}", word),
                tier: VocabularyTier::Basic,
                mass: 10.0,
                tags: vec!["Offline".to_string()],
            });
        }
    }
    
    None
}
